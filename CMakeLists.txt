cmake_minimum_required(VERSION 3.16)
project(sf_overlay_engine CXX)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
  set_property(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS Debug Release RelWithDebInfo MinSizeRel)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SDK_INCLUDE "$ENV{HOME}/.sdk/include" CACHE PATH "OpenVR headers")
set(QJS_INCLUDE "$ENV{HOME}/Code/third_party/quickjs-ng" CACHE PATH "QuickJS headers")
set(QJS_LIB "$ENV{HOME}/Code/third_party/quickjs-ng/build/libqjs.a" CACHE FILEPATH "QuickJS library")
set(STEAMVR_BIN "/opt/steamvr/bin/linuxarm64" CACHE PATH "SteamVR binaries")
find_path(DRM_INCLUDE drm_fourcc.h PATH_SUFFIXES libdrm REQUIRED)
find_package(Freetype REQUIRED)
add_executable(sf-overlay-engine src/main.cpp src/model_asset.cpp src/controller_input.cpp src/frame_input.cpp src/panel.cpp)
target_include_directories(sf-overlay-engine PRIVATE ${SDK_INCLUDE} ${QJS_INCLUDE} ${DRM_INCLUDE})
target_link_directories(sf-overlay-engine PRIVATE ${STEAMVR_BIN})
target_link_libraries(sf-overlay-engine PRIVATE openvr_api ${QJS_LIB} Freetype::Freetype pthread EGL GLESv2 gbm drm m)
