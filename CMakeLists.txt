cmake_minimum_required(VERSION 3.20)
project(sf_overlay_engine CXX)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
  set_property(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS Debug Release RelWithDebInfo MinSizeRel)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(runtime-stats-regression EXCLUDE_FROM_ALL tests/runtime-stats-regression.cpp)
target_include_directories(runtime-stats-regression PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
add_executable(scene-lookup-regression EXCLUDE_FROM_ALL tests/scene-lookup-regression.cpp)
target_include_directories(scene-lookup-regression PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
include(cmake/Physics.cmake)
option(SF_BUILD_ENGINE "Build the OpenVR overlay executable" ON)
if(NOT SF_BUILD_ENGINE)
  return()
endif()
set(SDK_INCLUDE "$ENV{HOME}/.sdk/include" CACHE PATH "OpenVR headers")
set(QJS_INCLUDE "$ENV{HOME}/Code/third_party/quickjs-ng" CACHE PATH "QuickJS headers")
set(QJS_LIB "$ENV{HOME}/Code/third_party/quickjs-ng/build/libqjs.a" CACHE FILEPATH "QuickJS library")
set(STEAMVR_BIN "/opt/steamvr/bin/linuxarm64" CACHE PATH "SteamVR binaries")
find_path(DRM_INCLUDE drm_fourcc.h PATH_SUFFIXES libdrm REQUIRED)
find_package(Freetype REQUIRED)
add_executable(sf-overlay-engine src/main.cpp src/model_asset.cpp src/controller_input.cpp src/frame_input.cpp src/panel.cpp)
# Explicit regression target: uses the actual QuickJS loader without starting VR.
add_executable(lifecycle-script-regression EXCLUDE_FROM_ALL tests/lifecycle-script-regression.cpp src/model_asset.cpp src/controller_input.cpp src/frame_input.cpp src/panel.cpp)
add_executable(animation-regression EXCLUDE_FROM_ALL tests/animation-regression.cpp src/model_asset.cpp src/controller_input.cpp src/frame_input.cpp src/panel.cpp)
add_executable(examples-regression EXCLUDE_FROM_ALL tests/examples-regression.cpp src/model_asset.cpp src/controller_input.cpp src/frame_input.cpp src/panel.cpp)
add_executable(physics-api-regression EXCLUDE_FROM_ALL tests/physics-api-regression.cpp src/model_asset.cpp src/controller_input.cpp src/frame_input.cpp src/panel.cpp)
add_executable(render-occlusion-regression EXCLUDE_FROM_ALL tests/render-occlusion-regression.cpp src/model_asset.cpp src/controller_input.cpp src/frame_input.cpp src/panel.cpp)
add_executable(render-stress-benchmark EXCLUDE_FROM_ALL tests/render-stress-benchmark.cpp src/model_asset.cpp src/controller_input.cpp src/frame_input.cpp src/panel.cpp)
add_executable(projected-bounds-regression EXCLUDE_FROM_ALL tests/projected-bounds-regression.cpp src/model_asset.cpp src/controller_input.cpp src/frame_input.cpp src/panel.cpp)
add_executable(panel-submission-regression EXCLUDE_FROM_ALL tests/panel-submission-regression.cpp src/model_asset.cpp src/controller_input.cpp src/frame_input.cpp src/panel.cpp)
add_executable(shared-depth-prototype EXCLUDE_FROM_ALL tests/shared-depth-prototype.cpp src/model_asset.cpp src/controller_input.cpp src/frame_input.cpp src/panel.cpp)
foreach(engine_target shared-depth-prototype projected-bounds-regression panel-submission-regression render-stress-benchmark render-occlusion-regression sf-overlay-engine lifecycle-script-regression animation-regression examples-regression physics-api-regression)
  target_include_directories(${engine_target} PRIVATE ${SDK_INCLUDE} ${QJS_INCLUDE} ${DRM_INCLUDE})
  target_link_directories(${engine_target} PRIVATE ${STEAMVR_BIN})
  target_link_libraries(${engine_target} PRIVATE sf-physics openvr_api ${QJS_LIB} Freetype::Freetype pthread EGL GLESv2 gbm drm m)
endforeach()
# Assertions are the regression checks even in a Release build.
target_compile_options(lifecycle-script-regression PRIVATE -UNDEBUG)

target_compile_options(animation-regression PRIVATE -UNDEBUG)

target_compile_options(examples-regression PRIVATE -UNDEBUG)

target_compile_options(physics-api-regression PRIVATE -UNDEBUG)

target_compile_options(render-occlusion-regression PRIVATE -UNDEBUG)

target_compile_options(render-stress-benchmark PRIVATE -UNDEBUG)
